## Exemplos Avançados
### Uso no Vue

<CodeExample 
  examples={{
    javascript: `<template>
  <div>
    <button class="alc-button" @click="refreshData">Adicionar Dados</button>
    <alc-datatable
      ref="datatable"
      :data="tableData"
      :options="datatableOptions"
      table-id="my-datatable"
    ></alc-datatable>
  </div>
</template>

<script>
export default {
  data() {
    return {
      tableData: [],
      datatableOptions: {
        columns: [
          { title: 'ID', data: 'id' },
          { title: 'Nome', data: 'name' },
          { title: 'Email', data: 'email' },
        ],
        stateSave: true,
      },
    };
  },
  mounted() {
    this.fetchData();
  },
  methods: {
    async fetchData() {
      // Simula uma chamada de API para obter os dados iniciais
      this.tableData = await new Promise((resolve) => {
        setTimeout(() => {
          resolve([
            { id: 1, name: 'João Silva', email: 'joao@example.com' },
            { id: 2, name: 'Maria Souza', email: 'maria@example.com' },
            { id: 3, name: 'Pedro Santos', email: 'pedro@example.com' },
          ]);
        }, 1000);
      });
    },
    async newData() {
      // Simula uma chamada de API para obter novos dados
      const newEntries = await new Promise((resolve) => {
        setTimeout(() => {
          resolve([
            { id: 4, name: 'José Oliveira', email: 'jose@example.com' },
            { id: 5, name: 'Ana Pereira', email: 'ana@example.com' },
          ]);
        }, 1000);
      });
      // Atualiza o array reativo tableData
      this.tableData = [...this.tableData, ...newEntries];
    },
    async refreshData() {
      // Adiciona novos dados e atualiza a tabela
      await this.newData();
      const datatable = await this.$refs.datatable.getApi();
      datatable.clear().draw();
      datatable.rows.add(this.tableData).draw();
    },
  },
};
</script>`,
    typescript: `<template>
  <div>
    <button class="alc-button" @click="refreshData">Adicionar Dados</button>
    <alc-datatable
      ref="datatable"
      :data="tableData"
      :options="datatableOptions"
      table-id="my-datatable"
    ></alc-datatable>
  </div>
</template>

<script lang="ts">
import { defineComponent, ref, onMounted } from 'vue';

export default defineComponent({
  name: 'DatatableExample',
  setup() {
    const tableData = ref<Array<{ id: number; name: string; email: string }>>([]);
    const datatableOptions = {
      columns: [
        { title: 'ID', data: 'id' },
        { title: 'Nome', data: 'name' },
        { title: 'Email', data: 'email' },
      ],
      stateSave: true,
    };
    const datatable = ref<any>(null);

    const fetchData = async () => {
      // Simula uma chamada de API para obter os dados iniciais
      tableData.value = await new Promise((resolve) => {
        setTimeout(() => {
          resolve([
            { id: 1, name: 'João Silva', email: 'joao@example.com' },
            { id: 2, name: 'Maria Souza', email: 'maria@example.com' },
            { id: 3, name: 'Pedro Santos', email: 'pedro@example.com' },
          ]);
        }, 1000);
      });
    };

    const newData = async () => {
      // Simula uma chamada de API para obter novos dados
      const newEntries = await new Promise((resolve) => {
        setTimeout(() => {
          resolve([
            { id: 4, name: 'José Oliveira', email: 'jose@example.com' },
            { id: 5, name: 'Ana Pereira', email: 'ana@example.com' },
          ]);
        }, 1000);
      });
      // Atualiza o array reativo tableData
      tableData.value = [...tableData.value, ...newEntries];
    };

    const refreshData = async () => {
      await newData();
      const dt = await datatable.value.getApi();
      dt.clear().draw();
      dt.rows.add(tableData.value).draw();
    };

    onMounted(() => {
      fetchData();
    });

    return {
      tableData,
      datatableOptions,
      datatable,
      refreshData,
    };
  },
});
</script>`
  }}
